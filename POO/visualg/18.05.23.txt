exemplo 3:
algoritmo "cinema"
var
   polt: vetor [1..100] de inteiro
   i,j,cont, menu: inteiro
inicio
   escreval
   escreval("          CINEMA SENAI        ")
   escreval
   para i de 1 ate 100 faca
        polt[i] <- i
   fimpara
   menu <- 0
   enquanto menu <> 3 faca
   escreval
   escreval("Escolha: 1 - Mapa das Cadeiras")
   escreval("         2 - Venda de Ingressos")
   escreval("         3 - Sair")
   escreval
   leia(menu)
   escreval
   limpatela
   se menu = 1 entao
      cont <- 1
           para j de 1 ate 20 faca
                escreval (polt[cont]:3, "   ", polt[cont+1]:3, "   ", polt[cont+2]:3, "    ", polt[cont+3]:3, "   ", polt[cont+4]:3)
                cont <- cont+5
           fimpara
      senao
           se menu = 2 entao
              escreva ("Qual a poltrona?  ")
              leia (i)
                   se polt[i] = 0 entao
                      escreval ("Desculpe, Poltrona ocupada. Verifique outro assento.")
                      cont <- 1
                           para j de 1 ate 20 faca
                                escreval (polt[cont]:3, "   ", polt[cont+1]:3, "   ", polt[cont+2]:3, "    ", polt[cont+3]:3, "   ", polt[cont+4]:3)
                                cont <- cont+5
                           fimpara
                      senao
                        polt[i] <- 0
                        escreval ("Parabéns a Poltrona", i, " foi vendida")
                   fimse
              senao
                   escreval ("Muito obrigado por utilizar nosso sistema")
           fimse
   fimse
   fimenquanto
